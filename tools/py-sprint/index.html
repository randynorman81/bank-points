1. Variables & Math Operators
name = "Ada"      # str
age = 17          # int
gpa = 3.95        # float
ready = True      # bool (True/False)
x += 5            # add 5 to x
17 // 5           # 3 (int division)
17 % 5            # 2 (MOD remainder)
2 ** 3            # 8 (exponent)

        
2. Lists & Procedures
nums = [10, 20, 30]
nums[0]           # 10 (0-indexed!)
nums[-1]          # 30 (last item)
len(nums)         # 3
nums.append(40)   # add to end
nums.insert(1,15) # insert at index 1
x = nums.pop(0)   # remove & return idx 0

        
3. Conditionals & Booleans
if score >= 90:
    grade = "A"
elif score >= 80:
    grade = "B"
else:
    grade = "C"
# Logic: and, or, not, ==, !=

        
4. Loops, Functions & Random
for i in range(1, 6): # 1,2,3,4,5
    print(i)

def add(a, b):
    return a + b

import random
r = random.randint(1, 6) # inclusive
